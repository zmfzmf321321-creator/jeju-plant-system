// 모델 로딩 완료 시 파트별 재질 색상 자동 채색 viewer.addEventListener('load', () => { const materials = viewer.model.materials; materials.forEach((mat, idx) => { const name = (mat.name || '').toLowerCase(); // 굴뚝 및 배기 덕트 라인: 안전 오렌지 / 메탈릭 브라운 if (name.includes('stack') || name.includes('chimney') || idx % 4 === 1) { mat.pbrMetallicRoughness.setBaseColorFactor([0.92, 0.45, 0.15, 1.0]); // 진한 주황/메탈 mat.pbrMetallicRoughness.setRoughnessFactor(0.4); mat.pbrMetallicRoughness.setMetallicFactor(0.6); } // 보일러 본체 및 대형 케이싱: 중후한 딥 플랜트 블루 else if (name.includes('boiler') || name.includes('body') || idx % 4 === 2) { mat.pbrMetallicRoughness.setBaseColorFactor([0.15, 0.35, 0.65, 1.0]); // 플랜트 블루 mat.pbrMetallicRoughness.setRoughnessFactor(0.3); mat.pbrMetallicRoughness.setMetallicFactor(0.4); } // 바닥 프레임 및 지지 구조물: 다크 차콜 / 스틸 그레이 else if (idx % 4 === 3) { mat.pbrMetallicRoughness.setBaseColorFactor([0.22, 0.26, 0.32, 1.0]); // 다크 메탈 mat.pbrMetallicRoughness.setRoughnessFactor(0.7); mat.pbrMetallicRoughness.setMetallicFactor(0.8); } // 기타 덕트 및 부속 배관: 밝은 쿨 그레이 else { mat.pbrMetallicRoughness.setBaseColorFactor([0.75, 0.82, 0.88, 1.0]); // 스틸 실버 mat.pbrMetallicRoughness.setRoughnessFactor(0.3); mat.pbrMetallicRoughness.setMetallicFactor(0.5); } }); });
